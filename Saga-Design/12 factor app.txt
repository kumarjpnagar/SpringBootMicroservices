4. Backing Services - Treat Backing Services as Attached Resources
java
// OrderServiceApplication.java
package com.example.order;

import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.cloud.client.discovery.EnableDiscoveryClient;
import org.springframework.cloud.openfeign.EnableFeignClients;
import org.springframework.kafka.annotation.EnableKafka;

/**
 * Order Service - Implements 12-Factor Factor IV: Backing Services
 * 
 * Backing services (database, Kafka, payment API) are treated as attached resources:
 * - All services are accessed via URLs/credentials from environment
 * - Services can be swapped without code changes
 * - Local and production services use same code, different configuration
 */
@SpringBootApplication
@EnableDiscoveryClient
@EnableFeignClients
@EnableKafka
public class OrderServiceApplication {
    
    public static void main(String[] args) {
        SpringApplication.run(OrderServiceApplication.class, args);
    }
}

/**
 * Database Configuration - Externalized via JPA
 * Connection details come from environment, not hardcoded
 */
@Configuration
public class DatabaseConfig {
    
    @Value("${spring.datasource.url}")
    private String databaseUrl; // From environment
    
    @Value("${spring.datasource.username}")
    private String username; // From environment
    
    @Value("${spring.datasource.password}")
    private String password; // From environment
    
    @Bean
    public DataSource dataSource() {
        // DataSource is configured from externalized configuration
        // Can be swapped between PostgreSQL, MySQL, etc. without code changes
        return DataSourceBuilder.create()
                .url(databaseUrl)
                .username(username)
                .password(password)
                .build();
    }
}

/**
 * Kafka Configuration - Backing service for messaging
 */
@Configuration
public class KafkaConfig {
    
    @Value("${spring.kafka.bootstrap-servers}")
    private String bootstrapServers; // From environment
    
    @Bean
    public KafkaTemplate<String, Object> kafkaTemplate() {
        // Kafka connection is externalized
        // Can point to local Kafka, Confluent Cloud, or MSK without code changes
        return new KafkaTemplate<>(producerFactory());
    }
    
    private ProducerFactory<String, Object> producerFactory() {
        Map<String, Object> config = new HashMap<>();
        config.put(ProducerConfig.BOOTSTRAP_SERVERS_CONFIG, bootstrapServers);
        config.put(ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG, StringSerializer.class);
        config.put(ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG, JsonSerializer.class);
        return new DefaultKafkaProducerFactory<>(config);
    }
}

/**
 * Payment Service Client - External API as backing service
 */
@FeignClient(name = "payment-service", configuration = FeignConfig.class)
public interface PaymentServiceClient {
    
    @PostMapping("/api/v1/payments")
    PaymentResponse processPayment(@RequestBody PaymentRequest request);
    
    /**
     * Feign configuration - Service discovery and timeouts from environment
     */
    @Configuration
    class FeignConfig {
        
        @Value("${service.payment.timeout:5000}")
        private int timeout; // From environment
        
        @Bean
        public Request.Options options() {
            // Timeout values can be adjusted per environment
            return new Request.Options(
                Duration.ofMillis(timeout),  // Connect timeout
                Duration.ofMillis(timeout)   // Read timeout
            );
        }
    }
}

6. Processes - Execute the App as One or More Stateless Processes
java
// OrderService.java - Stateless process implementation
package com.example.order.service;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;

/**
 * Order Service Implementation - Implements 12-Factor Factor VI: Processes
 * 
 * Key principles implemented:
 * 1. Stateless - No session affinity required
 * 2. Process-level isolation - State stored in backing services (DB)
 * 3. Horizontal scalability - Multiple instances can run simultaneously
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class OrderService {
    
    private final OrderRepository orderRepository;
    private final KafkaTemplate<String, Object> kafkaTemplate;
    private final PaymentServiceClient paymentClient;
    
    /**
     * Create order - Stateless operation
     * All state is persisted to database, not stored in memory
     * Multiple instances can process orders concurrently
     */
    @Transactional
    public Order createOrder(CreateOrderRequest request) {
        log.info("Creating order for user: {}", request.getUserId());
        
        // Create order entity
        Order order = Order.builder()
            .userId(request.getUserId())
            .items(request.getItems())
            .totalAmount(calculateTotal(request.getItems()))
            .status(OrderStatus.PENDING)
            .createdAt(Instant.now())
            .build();
        
        // Persist to database - External backing service
        Order savedOrder = orderRepository.save(order);
        log.info("Order created with ID: {}", savedOrder.getId());
        
        // Send event - No local state retention
        publishOrderCreatedEvent(savedOrder);
        
        return savedOrder;
    }
    
    /**
     * Process payment - Demonstrates stateless processing
     * Session data is not stored locally, uses external services
     */
    @Transactional
    public Order processPayment(Long orderId, PaymentDetails paymentDetails) {
        // Retrieve from DB - No local cache
        Order order = orderRepository.findById(orderId)
            .orElseThrow(() -> new OrderNotFoundException(orderId));
        
        try {
            // Call external payment service - No local state
            PaymentResponse paymentResponse = paymentClient.processPayment(
                PaymentRequest.builder()
                    .orderId(order.getId())
                    .amount(order.getTotalAmount())
                    .paymentDetails(paymentDetails)
                    .build()
            );
            
            // Update state in DB only
            order.setPaymentId(paymentResponse.getPaymentId());
            order.setStatus(OrderStatus.PAID);
            order.setUpdatedAt(Instant.now());
            
            Order updatedOrder = orderRepository.save(order);
            
            // Publish event - Asynchronous, stateless
            publishOrderPaidEvent(updatedOrder);
            
            return updatedOrder;
            
        } catch (Exception e) {
            log.error("Payment failed for order: {}", orderId, e);
            order.setStatus(OrderStatus.PAYMENT_FAILED);
            order.setErrorMessage(e.getMessage());
            return orderRepository.save(order);
        }
    }
    
    /**
     * Publish event - Stateless message production
     * No memory of published messages kept locally
     */
    private void publishOrderCreatedEvent(Order order) {
        OrderCreatedEvent event = OrderCreatedEvent.builder()
            .orderId(order.getId())
            .userId(order.getUserId())
            .totalAmount(order.getTotalAmount())
            .timestamp(Instant.now())
            .build();
        
        // Send to Kafka - External messaging service
        kafkaTemplate.send("order-created", order.getUserId(), event);
        log.debug("Order created event published: {}", event.getOrderId());
    }
    
    private void publishOrderPaidEvent(Order order) {
        OrderPaidEvent event = OrderPaidEvent.builder()
            .orderId(order.getId())
            .paymentId(order.getPaymentId())
            .amount(order.getTotalAmount())
            .timestamp(Instant.now())
            .build();
        
        kafkaTemplate.send("order-paid", order.getUserId(), event);
        log.debug("Order paid event published: {}", event.getOrderId());
    }
    
    /**
     * Calculate total - Pure function, no side effects
     */
    private BigDecimal calculateTotal(List<OrderItem> items) {
        return items.stream()
            .map(item -> item.getPrice().multiply(BigDecimal.valueOf(item.getQuantity())))
            .reduce(BigDecimal.ZERO, BigDecimal::add);
    }
}

// WebServerConfig.java
package com.example.order.config;

import org.springframework.boot.web.embedded.tomcat.TomcatServletWebServerFactory;
import org.springframework.boot.web.server.WebServerFactoryCustomizer;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

/**
 * Port Binding Configuration - Implements 12-Factor Factor VII: Port Binding
 * 
 * The app is self-contained and exports HTTP as a service by binding to a port.
 * Port is configured via environment variable for cloud-native deployment.
 */
@Configuration
public class WebServerConfig {
    
    /**
     * Configure web server to bind to port from environment
     * This makes the app completely self-contained - no separate web server needed
     */
    @Bean
    public WebServerFactoryCustomizer<TomcatServletWebServerFactory> webServerFactoryCustomizer() {
        return factory -> {
            // Port is provided by environment (${PORT:8080})
            // In cloud platforms (Heroku, Cloud Foundry, Kubernetes), PORT env var is provided
            factory.setPort(getPortFromEnvironment());
            
            // Bind to all interfaces - Allows cloud load balancers to connect
            factory.setAddress(null); // null = 0.0.0.0 (all interfaces)
            
            // Customize connector
            factory.addConnectorCustomizers(connector -> {
                connector.setProperty("bindOnInit", "false");
                connector.setProperty("acceptorThreadCount", "2");
            });
        };
    }
    
    private int getPortFromEnvironment() {
        String port = System.getenv("PORT");
        if (port != null && !port.isEmpty()) {
            return Integer.parseInt(port);
        }
        // Fallback to application.properties value
        return 8080;
    }
}

/**
 * Application entry point with port binding
 */
@SpringBootApplication
public class OrderServiceApplication {
    
    public static void main(String[] args) {
        // Spring Boot automatically binds to port specified in environment
        // No external web server required
        SpringApplication app = new SpringApplication(OrderServiceApplication.class);
        
        // Set default properties from environment
        Properties defaults = new Properties();
        defaults.put("server.port", System.getenv().getOrDefault("PORT", "8080"));
        defaults.put("server.address", "0.0.0.0");
        
        app.setDefaultProperties(defaults);
        app.run(args);
        
        log.info("Order Service started and bound to port: {}", 
                 System.getenv().getOrDefault("PORT", "8080"));
    }
}

// GracefulShutdownConfig.java
package com.example.order.config;

import org.springframework.context.event.ContextClosedEvent;
import org.springframework.context.event.EventListener;
import org.springframework.stereotype.Component;
import lombok.extern.slf4j.Slf4j;

import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

/**
 * Disposability Configuration - Implements 12-Factor Factor IX: Disposability
 * 
 * Processes can be started or stopped at any time.
 * Key features:
 * - Fast startup (seconds, not minutes)
 * - Graceful shutdown (complete ongoing requests)
 * - Robust against sudden death
 */
@Component
@Slf4j
public class GracefulShutdownConfig {
    
    private final ExecutorService taskExecutor = Executors.newFixedThreadPool(10);
    
    /**
     * Handle graceful shutdown on application context close
     * Waits for in-flight requests to complete before shutdown
     */
    @EventListener
    public void onApplicationEvent(ContextClosedEvent event) {
        log.info("Received shutdown signal. Starting graceful shutdown...");
        
        // 1. Stop accepting new requests
        shutdownWebServer();
        
        // 2. Complete existing requests
        waitForRequestsToComplete();
        
        // 3. Clean up resources
        cleanupResources();
        
        log.info("Graceful shutdown completed.");
    }
    
    private void shutdownWebServer() {
        // Tomcat will stop accepting new connections
        log.info("Stopping HTTP server from accepting new connections");
        // Implementation depends on web server
    }
    
    private void waitForRequestsToComplete() {
        log.info("Waiting for in-flight requests to complete...");
        
        try {
            // Give existing requests time to complete
            taskExecutor.shutdown();
            
            boolean terminated = taskExecutor.awaitTermination(30, TimeUnit.SECONDS);
            
            if (!terminated) {
                log.warn("Some tasks did not complete within timeout. Forcing shutdown.");
                taskExecutor.shutdownNow();
            }
            
        } catch (InterruptedException e) {
            log.error("Interrupted while waiting for shutdown", e);
            Thread.currentThread().interrupt();
        }
    }
    
    private void cleanupResources() {
        // Close database connections
        // Close Kafka producers/consumers
        // Release file handles
        log.info("Resources cleaned up successfully");
    }
    
    /**
     * Fast startup verification
     * Spring Boot actuators expose startup metrics
     */
    @Component
    public class StartupMetrics {
        
        private final long startupTime = System.currentTimeMillis();
        
        @EventListener(ApplicationReadyEvent.class)
        public void onApplicationReady() {
            long startupDuration = System.currentTimeMillis() - startupTime;
            log.info("Application started in {} ms", startupDuration);
            
            if (startupDuration > 10000) { // More than 10 seconds
                log.warn("Startup time exceeds recommended threshold: {} ms", startupDuration);
            }
        }
    }
}

package com.example.order.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.boot.logging.LoggingSystem;
import ch.qos.logback.classic.AsyncAppender;
import ch.qos.logback.classic.LoggerContext;
import net.logstash.logback.appender.LogstashTcpSocketAppender;
import net.logstash.logback.encoder.LogstashEncoder;

/**
 * Logging Configuration - Implements 12-Factor Factor XI: Logs
 * 
 * Logs are treated as event streams:
 * - Write to stdout/stderr only
 * - No log file management in app
 * - Execution environment handles log routing
 */
@Configuration
public class LoggingConfig {
    
    @Bean
    public LoggingSystem loggingSystem() {
        return LoggingSystem.get(ClassLoader.getSystemClassLoader());
    }
    
    /**
     * Configure logging to write only to stdout
     * No file appenders - Let environment handle aggregation
     */
    @Configuration
    public static class LogbackConfig {
        
        @Value("${logging.json.enabled:false}")
        private boolean jsonLoggingEnabled;
        
        @Value("${logging.console.enabled:true}")
        private boolean consoleLoggingEnabled;
        
        @Bean
        public LoggerContext loggerContext() {
            LoggerContext context = (LoggerContext) LoggerFactory.getILoggerFactory();
            
            // Clear existing appenders - No file appenders
            context.reset();
            
            if (consoleLoggingEnabled) {
                if (jsonLoggingEnabled) {
                    // JSON format for structured logging
                    addJsonConsoleAppender(context);
                } else {
                    // Human-readable format for development
                    addHumanReadableConsoleAppender(context);
                }
            }
            
            // No file appenders - Logs go to stdout only
            
            // Set root logger level from environment
            String logLevel = System.getenv("LOG_LEVEL");
            if (logLevel != null) {
                context.getLogger(Logger.ROOT_LOGGER_NAME).setLevel(
                    Level.toLevel(logLevel, Level.INFO)
                );
            }
            
            return context;
        }
        
        private void addHumanReadableConsoleAppender(LoggerContext context) {
            ConsoleAppender<ILoggingEvent> consoleAppender = new ConsoleAppender<>();
            consoleAppender.setContext(context);
            consoleAppender.setName("STDOUT");
            
            PatternLayoutEncoder encoder = new PatternLayoutEncoder();
            encoder.setContext(context);
            encoder.setPattern("%d{yyyy-MM-dd HH:mm:ss.SSS} [%thread] %-5level %logger{36} - %msg%n");
            encoder.start();
            
            consoleAppender.setEncoder(encoder);
            consoleAppender.start();
            
            context.getLogger(Logger.ROOT_LOGGER_NAME).addAppender(consoleAppender);
        }
        
        private void addJsonConsoleAppender(LoggerContext context) {
            ConsoleAppender<ILoggingEvent> consoleAppender = new ConsoleAppender<>();
            consoleAppender.setContext(context);
            consoleAppender.setName("STDOUT-JSON");
            
            LogstashEncoder encoder = new LogstashEncoder();
            encoder.setContext(context);
            
            // Add service context
            Map<String, String> customFields = new HashMap<>();
            customFields.put("service", "order-service");
            customFields.put("version", System.getenv("RELEASE_VERSION"));
            encoder.setCustomFields(new ObjectMapper().writeValueAsString(customFields));
            
            encoder.start();
            
            consoleAppender.setEncoder(encoder);
            consoleAppender.start();
            
            context.getLogger(Logger.ROOT_LOGGER_NAME).addAppender(consoleAppender);
        }
    }
}

package com.example.order.admin;

import org.springframework.boot.CommandLineRunner;
import org.springframework.context.annotation.Profile;
import org.springframework.stereotype.Component;
import lombok.extern.slf4j.Slf4j;

/**
 * Database Migration - Implements 12-Factor Factor XII: Admin Processes
 * 
 * Admin processes are run in the same environment as the app:
 * - One-off migrations
 * - Data fixes
 * - Reporting scripts
 * 
 * These run against production databases using the same codebase and config
 */
@Component
@Profile("migration")  // Activated only for migration tasks
@Slf4j
public class DatabaseMigrationRunner implements CommandLineRunner {
    
    private final DataSource dataSource;
    private final MigrationService migrationService;
    
    public DatabaseMigrationRunner(DataSource dataSource, MigrationService migrationService) {
        this.dataSource = dataSource;
        this.migrationService = migrationService;
    }
    
    @Override
    public void run(String... args) throws Exception {
        log.info("Running database migration as one-off process");
        
        try {
            // Connect to production database using same configuration
            Connection connection = dataSource.getConnection();
            log.info("Connected to database: {}", connection.getMetaData().getURL());
            
            // Run migrations
            migrationService.runMigrations();
            
            log.info("Database migration completed successfully");
            
        } catch (Exception e) {
            log.error("Migration failed", e);
            System.exit(1);  // Exit with error code
        }
        
        // Exit after completion
        System.exit(0);
    }
}

@Profile("fix-order-data")

        log.info("Found {} orders requiring fix", problematicOrders.size());

        for (Order order : problematicOrders) {

            }
        }
        
        log.info("Fixed {} orders successfully", fixed);
    }

        }
