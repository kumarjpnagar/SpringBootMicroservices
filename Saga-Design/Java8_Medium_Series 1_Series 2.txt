18. Find the nth Fibonacci number using streams.

int n=3; //3rd fibonacci Number
  int nthFibNo=  generateFibnacciStream()
                 .skip(n-1)
                 .findFirst()
                 .orElseThrow(()->new IllegalArgumentException("Empty List"));
  
  System.out.println(nthFibNo);
}
 
 
 private static Stream<Integer> generateFibnacciStream(){
  
  return Stream.iterate(new int[] {0,1}, num->new int[]{num[1],num[0]+num[1]})
      .map(num->num[0]);
 }

 }
 
 private static <T extends Comparable<? super T>> boolean isSortedAscending(List<T> list)
 {
  
  return IntStream.range(0, list.size()-1)
                  .allMatch(i-> list.get(i)
                                    .compareTo(list.get(i+1))<=0);
  
  
 }

//output: true
//        false

/* In Java, you cannot directly remove elements from a list 
while iterating over it using streams. 
Streams in Java are designed to work with immutable data structures, 
and removing elements from a list while iterating using streams would 
violate this principle. */

List<Integer> numbers = new ArrayList<>(List.of(1, 2, 3, 4, 5));

        Iterator<Integer> iterator = numbers.iterator();
        while (iterator.hasNext()) {
            int number = iterator.next();
            if (number % 2 == 0) {
                iterator.remove();
            }
        }

        System.out.println("List after removing even numbers: " + numbers);

 private static List<Integer> generateAlternateList(List<Integer> list1, List<Integer> list2)
 {
  if(list1.size()!=list2.size())
   throw new IllegalArgumentException("List should must be equal in size");
  
  List<Integer> lst=IntStream.range(0, list1.size())
                      .mapToObj(i->Arrays.asList(list1.get(i), list2.get(i)))
                      .flatMap(List::stream)
                      .collect( Collectors.toList()); 
  return lst;
 }

19. Find the median of a list of numbers using streams.

       int mid=list1.size()/2;
        
       Double median= list1.stream()
                           .sorted()
                           .skip(mid-(isEven?1:0))
                           .limit(isEven?2:1)
                           .mapToInt(Integer::intValue)
                           .average()
                           .orElse(Double.NaN);

 public static boolean isPrime(int number)
 {
  if(number<=1)
   return false;
  
  else
  {
   for(int i=0;i<Math.sqrt(number);i++)
   {
    if(number%2==0)
     return false;
 
   }
   
   return true;
  }
 }
