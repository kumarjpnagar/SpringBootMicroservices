// --- Employee Class Definition (Used in Many Examples) ---
class Employee {
    int id;
    String name;
    String department;
    double salary;
    int age;

    // Constructor for Employee with age
    public Employee(int id, String name, String department, double salary, int age) {
        this.id = id;
        this.name = name;
        this.department = department;
        this.salary = salary;
        this.age = age;
    }

    // Constructor for Employee without age (for simpler examples)
    public Employee(int id, String name, String department, double salary) {
        this.id = id;
        this.name = name;
        this.department = department;
        this.salary = salary;
        this.age = 0; // Default age
    }

    // Getters
    public int getId() { return id; }
    public String getName() { return name; }
    public String getDepartment() { return department; }
    public double getSalary() { return salary; }
    public int getAge() { return age; }

    @Override
    public String toString() {
        return String.format("Employee{id=%d, name='%s', dept='%s', salary=%.2f, age=%d}", 
                             id, name, department, salary, age);
    }
}

    public static void findDuplicates() {
        System.out.println("\n--- 1. Find Duplicate Elements ---");
        List<Integer> list = Arrays.asList(1, 2, 3, 2, 4, 5, 1);
        Set<Integer> seen = new HashSet<>();
        Set<Integer> duplicates = list.stream()
                .filter(n -> !seen.add(n))
                .collect(Collectors.toSet());
        System.out.println("Duplicates in " + list + ": " + duplicates);
    }

    // --- 35. Filter Even Numbers from Mixed List using Predicate ---
    public static void evenFromMixedList() {
        System.out.println("\n--- 35. Even Numbers from Mixed List ---");
        List<String> input = Arrays.asList("1","2","3","4","abc","5");
        Predicate<String> isNumeric = s -> {
            try {
                Integer.parseInt(s);
                return true;
            } catch (NumberFormatException e) {
                return false;
            }
        };
        List<Integer> evenNumbers = input.stream()
                .filter(isNumeric)
                .map(Integer::parseInt)
                .filter(n -> n % 2 == 0)
                .collect(Collectors.toList());
        System.out.println("Even numbers from " + input + ": " + evenNumbers);
    }

    // --- 36. Sort HashMap by Value ---
    public static void sortHashMapByValue() {
        System.out.println("\n--- 36. Sort HashMap by Value ---");
        Map<Integer, String> map = new HashMap<>();
        map.put(1, "x");
        map.put(2, "s");
        map.put(3, "z");
        System.out.println("Original Map: " + map);
        Map<Integer, String> sortedMap = map.entrySet().stream()
                .sorted(Map.Entry.comparingByValue())
                .collect(Collectors.toMap(
                        Map.Entry::getKey,
                        Map.Entry::getValue,
                        (v1, v2) -> v1,
                        LinkedHashMap::new
                ));
        System.out.println("Sorted Map (by value): " + sortedMap);
    }

    // --- 40. Check if Two Strings are Anagrams ---
    public static void checkAnagrams() {

    // --- 41. Check Palindrome using Streams ---
    public static void checkPalindrome() {
        System.out.println("\n--- 41. Check Palindrome using Streams ---");
        String str = "radar";
        boolean isPalindrome = IntStream.range(0, str.length() / 2)
                .allMatch(i -> str.charAt(i) == str.charAt(str.length() - 1 - i));
        System.out.println("'" + str + "' is a palindrome: " + isPalindrome);
    }

    // --- 42. Replace Character in String ---
    public static void replaceCharacter() {
        System.out.println("\n--- 42. Replace Character in String ---");
        String str = "hello world";
        char oldChar = 'o';
        char newChar = 'x';
        String replaced = str.chars()
                .map(c -> c == oldChar ? newChar : c)
                .mapToObj(c -> String.valueOf((char) c))
                .collect(Collectors.joining());
        System.out.println("Original: '" + str + "' -> Replaced '" + oldChar + "' with '" + newChar + "': '" + replaced + "'");
    }

    // --- 45. Thread-Safe Rate Limiter (Conceptual) ---
    // Note: This requires an external library (bucket4j). 
    // This is a simplified version showing the core concept using Java's atomic package.
    public static void rateLimiterConcept() {
        System.out.println("\n--- 45. Thread-Safe Rate Limiter Concept (Simplified) ---");
        // This is a conceptual implementation using AtomicInteger for simplicity.
        // A real implementation would use a token bucket algorithm (like bucket4j).
        java.util.concurrent.atomic.AtomicInteger counter = new java.util.concurrent.atomic.AtomicInteger(0);
        int maxRequests = 10;
        long windowMillis = 60000; // 60 seconds

        // Simulating requests in a single-threaded manner for demonstration
        System.out.println("Simulating requests... (conceptual)");
        for (int i = 0; i < 12; i++) {
            if (counter.incrementAndGet() <= maxRequests) {
                System.out.println("Request " + (i+1) + " allowed.");
            } else {
                System.out.println("Request " + (i+1) + " denied. Rate limit exceeded.");
            }
            // In a real scenario, we'd reset the counter after the time window.
            // This is a simplistic and non-thread-safe reset mechanism.
            if ((i+1) == maxRequests) {
                System.out.println("--- Resetting counter for the next window (conceptual) ---");
                counter.set(0);
            }
        }
        System.out.println("(Note: A production-grade rate limiter uses token bucket or leaky bucket algorithms.)");
    }
