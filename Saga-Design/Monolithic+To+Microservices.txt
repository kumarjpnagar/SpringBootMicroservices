1b. Add Service Mediation Layer (Event-based using in-memory queue)
java
// Prepare for Kafka later
public interface EventPublisher {
    void publish(String topic, Object event);
}

// In-memory implementation (monolith)
@Component
public class InMemoryEventPublisher implements EventPublisher {
    private final ApplicationEventPublisher springEvents;
    public void publish(String topic, Object event) {
        springEvents.publishEvent(new CustomEvent(topic, event));
    }
}

Why: Decouples modules inside monolith without distributed complexity.

Step-by-Step Implementation

1. Kafka Event Definition
java
// common/src/main/java/com/app/events/OrderCreatedEvent.java
public record OrderCreatedEvent(Long orderId, Long customerId, List<OrderLine> lines) {}

2. Order Service Producer
java
@Service
public class OrderEventPublisher {
    @Autowired private KafkaTemplate<String, OrderCreatedEvent> kafkaTemplate;

    public void publishOrderCreated(Order order) {
        OrderCreatedEvent event = new OrderCreatedEvent(order.getId(), order.getCustomerId(), order.getLines());
        kafkaTemplate.send("order-events", event);
    }
}

3. Inventory Service Consumer
java
@Component
public class InventoryEventHandler {
    @KafkaListener(topics = "order-events", groupId = "inventory-group")
    public void handleOrderCreated(OrderCreatedEvent event) {
        // check stock
        boolean available = checkStock(event.lines());
        if (!available) {
            // send out-of-stock event back
            kafkaTemplate.send("inventory-events", new OutOfStockEvent(event.orderId()));
        }
    }
}

    private final ApplicationEventPublisher publisher;

    public InMemoryEventPublisher(ApplicationEventPublisher publisher) {
        this.publisher = publisher;
    }

    public void publish(String topic, Object event) {
        publisher.publishEvent(event);
    }
}
