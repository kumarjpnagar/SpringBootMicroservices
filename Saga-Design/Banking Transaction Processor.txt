

// BankAccount.java
package com.banking.model;

import java.math.BigDecimal;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.UUID;

public class BankAccount {
    private final String accountId;
    private BigDecimal balance;
    private final List<Transaction> transactionHistory;
    private final BigDecimal overdraftLimit;

    public BankAccount(String accountId, BigDecimal initialBalance) {
        this(accountId, initialBalance, BigDecimal.ZERO);
    }

    public BankAccount(String accountId, BigDecimal initialBalance, BigDecimal overdraftLimit) {
        validateAccountId(accountId);
        validateAmount(initialBalance);
        validateOverdraftLimit(overdraftLimit);
        
        this.accountId = accountId;
        this.balance = initialBalance;
        this.transactionHistory = new ArrayList<>();
        this.overdraftLimit = overdraftLimit;
        
        // Record initial deposit as a transaction
        if (initialBalance.compareTo(BigDecimal.ZERO) > 0) {
            this.transactionHistory.add(new Transaction(
                TransactionType.DEPOSIT,
                initialBalance,
                this.balance,
                "Initial deposit"
            ));
        }
    }

    public void deposit(BigDecimal amount) {
        deposit(amount, "Deposit");
    }

    public void deposit(BigDecimal amount, String description) {
        validateAmount(amount);
        
        BigDecimal newBalance = balance.add(amount);
        transactionHistory.add(new Transaction(
            TransactionType.DEPOSIT,
            amount,
            newBalance,
            description
        ));
        balance = newBalance;
    }

    public void withdraw(BigDecimal amount) {
        withdraw(amount, "Withdrawal");
    }

    public void withdraw(BigDecimal amount, String description) {
        validateAmount(amount);
        validateSufficientFunds(amount);
        
        BigDecimal newBalance = balance.subtract(amount);
        transactionHistory.add(new Transaction(
            TransactionType.WITHDRAWAL,
            amount,
            newBalance,
            description
        ));
        balance = newBalance;
    }

    public void transferTo(BankAccount targetAccount, BigDecimal amount) {
        transferTo(targetAccount, amount, "Transfer");
    }

    public void transferTo(BankAccount targetAccount, BigDecimal amount, String description) {
        if (targetAccount == null) {
            throw new IllegalArgumentException("Target account cannot be null");
        }
        if (this.equals(targetAccount)) {
            throw new IllegalArgumentException("Cannot transfer to the same account");
        }
        
        validateAmount(amount);
        validateSufficientFunds(amount);
        
        // Perform withdrawal from source
        BigDecimal newBalance = balance.subtract(amount);
        transactionHistory.add(new Transaction(
            TransactionType.TRANSFER_OUT,
            amount,
            newBalance,
            description + " to " + targetAccount.getAccountId()
        ));
        balance = newBalance;
        
        // Perform deposit to target
        targetAccount.deposit(amount, description + " from " + accountId);
    }

    public BigDecimal getBalance() {
        return balance;
    }

    public String getAccountId() {
        return accountId;
    }

    public List<Transaction> getTransactionHistory() {
        return Collections.unmodifiableList(transactionHistory);
    }

    public List<Transaction> getTransactionHistory(LocalDateTime from, LocalDateTime to) {
        if (from == null || to == null) {
            throw new IllegalArgumentException("Date range cannot be null");
        }
        if (from.isAfter(to)) {
            throw new IllegalArgumentException("From date must be before to date");
        }
        
        return transactionHistory.stream()
            .filter(t -> !t.getTimestamp().isBefore(from) && !t.getTimestamp().isAfter(to))
            .toList();
    }

    public BigDecimal getOverdraftLimit() {
        return overdraftLimit;
    }

    private void validateAmount(BigDecimal amount) {
        if (amount == null) {
            throw new IllegalArgumentException("Amount cannot be null");
        }
        if (amount.compareTo(BigDecimal.ZERO) <= 0) {
            throw new IllegalArgumentException("Amount must be positive");
        }
    }

    private void validateAccountId(String accountId) {
        if (accountId == null || accountId.trim().isEmpty()) {
            throw new IllegalArgumentException("Account ID cannot be null or empty");
        }
    }

    private void validateOverdraftLimit(BigDecimal overdraftLimit) {
        if (overdraftLimit == null || overdraftLimit.compareTo(BigDecimal.ZERO) < 0) {
            throw new IllegalArgumentException("Overdraft limit cannot be negative");
        }
    }

    private void validateSufficientFunds(BigDecimal amount) {
        BigDecimal availableBalance = balance.add(overdraftLimit);
        if (amount.compareTo(availableBalance) > 0) {
            throw new InsufficientFundsException(
                "Insufficient funds. Available: " + availableBalance + 
                ", Requested: " + amount + 
                (overdraftLimit.compareTo(BigDecimal.ZERO) > 0 ? 
                    " (including overdraft limit: " + overdraftLimit + ")" : "")
            );
        }
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        BankAccount that = (BankAccount) o;
        return accountId.equals(that.accountId);
    }

    @Override
    public int hashCode() {
        return accountId.hashCode();
    }
}




// Transaction.java
package com.banking.model;

import java.math.BigDecimal;
import java.time.LocalDateTime;
import java.util.UUID;

public class Transaction {
    private final String transactionId;
    private final TransactionType type;
    private final BigDecimal amount;
    private final BigDecimal balanceAfter;
    private final LocalDateTime timestamp;
    private final String description;

    public Transaction(TransactionType type, BigDecimal amount, BigDecimal balanceAfter, String description) {
        this.transactionId = UUID.randomUUID().toString();
        this.type = type;
        this.amount = amount;
        this.balanceAfter = balanceAfter;
        this.timestamp = LocalDateTime.now();
        this.description = description != null ? description : "";
    }

    public String getTransactionId() {
        return transactionId;
    }

    public TransactionType getType() {
        return type;
    }

    public BigDecimal getAmount() {
        return amount;
    }

    public BigDecimal getBalanceAfter() {
        return balanceAfter;
    }

    public LocalDateTime getTimestamp() {
        return timestamp;
    }

    public String getDescription() {
        return description;
    }

    @Override
    public String toString() {
        return "Transaction{" +
            "transactionId='" + transactionId + '\'' +
            ", type=" + type +
            ", amount=" + amount +
            ", balanceAfter=" + balanceAfter +
            ", timestamp=" + timestamp +
            ", description='" + description + '\'' +
            '}';
    }
}

// BankingService.java
package com.banking.service;

import com.banking.model.BankAccount;
import com.banking.model.Transaction;

import java.math.BigDecimal;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

public class BankingService {
    private final Map<String, BankAccount> accounts;

    public BankingService() {
        this.accounts = new ConcurrentHashMap<>();
    }

    public BankAccount createAccount(String accountId, BigDecimal initialBalance) {
        return createAccount(accountId, initialBalance, BigDecimal.ZERO);
    }

    public BankAccount createAccount(String accountId, BigDecimal initialBalance, BigDecimal overdraftLimit) {
        if (accounts.containsKey(accountId)) {
            throw new IllegalArgumentException("Account already exists with ID: " + accountId);
        }
        
        BankAccount account = new BankAccount(accountId, initialBalance, overdraftLimit);
        accounts.put(accountId, account);
        return account;
    }

    public BankAccount getAccount(String accountId) {
        BankAccount account = accounts.get(accountId);
        if (account == null) {
            throw new IllegalArgumentException("Account not found with ID: " + accountId);
        }
        return account;
    }

    public void deposit(String accountId, BigDecimal amount) {
        BankAccount account = getAccount(accountId);
        account.deposit(amount);
    }

    public void deposit(String accountId, BigDecimal amount, String description) {
        BankAccount account = getAccount(accountId);
        account.deposit(amount, description);
    }

    public void withdraw(String accountId, BigDecimal amount) {
        BankAccount account = getAccount(accountId);
        account.withdraw(amount);
    }

    public void withdraw(String accountId, BigDecimal amount, String description) {
        BankAccount account = getAccount(accountId);
        account.withdraw(amount, description);
    }

    public void transfer(String fromAccountId, String toAccountId, BigDecimal amount) {
        BankAccount fromAccount = getAccount(fromAccountId);
        BankAccount toAccount = getAccount(toAccountId);
        fromAccount.transferTo(toAccount, amount);
    }

    public void transfer(String fromAccountId, String toAccountId, BigDecimal amount, String description) {
        BankAccount fromAccount = getAccount(fromAccountId);
        BankAccount toAccount = getAccount(toAccountId);
        fromAccount.transferTo(toAccount, amount, description);
    }

    public BigDecimal getBalance(String accountId) {
        BankAccount account = getAccount(accountId);
        return account.getBalance();
    }

    public List<Transaction> getTransactionHistory(String accountId) {
        BankAccount account = getAccount(accountId);
        return account.getTransactionHistory();
    }

    public List<Transaction> getTransactionHistory(String accountId, LocalDateTime from, LocalDateTime to) {
        BankAccount account = getAccount(accountId);
        return account.getTransactionHistory(from, to);
    }

    public List<BankAccount> getAllAccounts() {
        return List.copyOf(accounts.values());
    }

    public boolean accountExists(String accountId) {
        return accounts.containsKey(accountId);
    }
}

            () -> bankingService.deposit("ACC001", BigDecimal.ZERO));

package com.banking;

import com.banking.model.BankAccount;

import java.math.BigDecimal;

public class Application {
    public static void main(String[] args) {
        BankingService bankingService = new BankingService();
        
        // Create accounts
        BankAccount account1 = bankingService.createAccount("ACC001", BigDecimal.valueOf(1000));
        BankAccount account2 = bankingService.createAccount("ACC002", BigDecimal.valueOf(500), BigDecimal.valueOf(200));
        
        System.out.println("Account 1 balance: " + account1.getBalance());

        System.out.println("Account 1 new balance: " + bankingService.getBalance("ACC001"));
        System.out.println("Account 2 new balance: " + bankingService.getBalance("ACC002"));
        
        // Get transaction history
        System.out.println("Account 1 transactions: " + bankingService.getTransactionHistory("ACC001"));
    }
}

# Banking Transaction Processor

A comprehensive banking transaction processing service that manages multiple accounts with deposit, withdrawal, and transfer operations.

## Features

- **Account Management**: Create accounts with unique IDs and initial balances
- **Transaction Operations**: 
  - Deposit funds with optional description
  - Withdraw funds with validation
  - Transfer between accounts
- **Validation**: 
  - Prevents overdrafts (with configurable overdraft limits)
  - Validates positive amounts
  - Ensures account existence
- **Transaction History**: Maintains immutable ledger with timestamps
- **Query APIs**: 
  - Get account balances
  - Get full transaction history
  - Get filtered history by date range

## Design Principles

- **Test-Driven Development**: Comprehensive unit tests with JUnit 5
- **Clean Code**: Clear naming, single responsibility, meaningful abstractions
- **Object-Oriented**: Proper encapsulation, immutability where appropriate
- **Edge Cases Handled**: Null checks, invalid amounts, overdraft protection

## Technology Stack

- Java 17
- JUnit 5 for testing
- Maven for build automation
